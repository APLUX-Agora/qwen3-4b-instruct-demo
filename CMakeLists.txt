cmake_minimum_required(VERSION 3.16)
project(qwen3_demo LANGUAGES CXX)
find_path(AIDGEN_INCLUDE_DIR aidlux/aidgen/aidgen.hpp REQUIRED)
find_library(AIDGEN_LIBRARY aidgen REQUIRED)
find_package(Threads REQUIRED)
add_executable(qwen3_demo examples/generate.cpp)
target_compile_features(qwen3_demo PRIVATE cxx_std_17)
target_include_directories(qwen3_demo PRIVATE ${AIDGEN_INCLUDE_DIR})
target_link_libraries(qwen3_demo PRIVATE ${AIDGEN_LIBRARY} Threads::Threads)
